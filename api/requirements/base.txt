# Shared dependencies for all environments.
#
# Django / django-environ / PyJWT are pinned to the EXACT versions used by
# the sibling `backend/` project (see backend/requirements/base.txt). This
# service calls `django.setup()` against that same project's code and
# database, so any drift here risks ORM/migration-state mismatches between
# the two processes.
Django==6.1.2
django-environ==0.14.0
PyJWT==2.15.1
# Not used by any of this service's own code. Required anyway: Django's
# settings loader imports `config` (the backend's project package) before
# `config.settings.<env>`, and `backend/config/__init__.py` unconditionally
# imports `backend/config/celery.py`, which imports `celery` at module level.
# Without this installed, `django.setup()` fails with
# `ModuleNotFoundError: No module named 'celery'` even though this service
# never creates or uses a Celery app itself. Pinned to the same version as
# backend/requirements/base.txt for consistency.
celery==5.6.3
# Same reason as celery above: `INSTALLED_APPS` (backend/config/settings/base.py)
# lists "corsheaders" and "rest_framework" as third-party apps, and Django's
# app registry (`django.apps.populate()`, called by `django.setup()`)
# imports every app in `INSTALLED_APPS` regardless of which settings module
# or environment is active — so these must be importable even though this
# service's own routes never use DRF or django-cors-headers directly (FastAPI
# has its own CORSMiddleware — see app/main.py). Pinned to the same versions
# as backend/requirements/base.txt.
django-cors-headers==4.9.0
djangorestframework==3.18.3
# Post.cover_image is an ImageField; Django's app loading / system checks
# touch Pillow for any installed model with an ImageField, so this must be
# installed too, even though this service never handles image uploads itself.
Pillow==13.0.0
# backend/config/settings/{local,production}.py both configure
# ENGINE=django.db.backends.postgresql, so any real ORM query this service
# runs through django.setup() needs a Postgres driver available here too
# (this is a separate process/venv from backend/, nothing is inherited from
# it). Pinned to the same version as backend/requirements/base.txt.
psycopg[binary]==3.3.6

# FastAPI side of this service.
fastapi==0.115.0
uvicorn[standard]==0.30.6
pydantic-settings==2.5.2
redis==8.1.0
